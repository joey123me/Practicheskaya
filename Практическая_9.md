```using System;

// Класс сущности: BatteryPack
public class BatteryPack
{
    // 4 закрытых поля (private) — состояние аккумулятора
    private double _capacityKwh;          // полная ёмкость в кВт·ч
    private int _chargePercent;           // текущий заряд в процентах (0–100)
    private double _temperatureC;         // температура батареи в °C
    private int _cycleCount;              // количество циклов перезарядки

    // Свойства с валидацией диапазонов
    public double CapacityKwh
    {
        get => _capacityKwh;
        set => _capacityKwh = value < 0 ? 0 : value;
    }

    public int ChargePercent
    {
        get => _chargePercent;
        set
        {
            if (value < 0) _chargePercent = 0;
            else if (value > 100) _chargePercent = 100;
            else _chargePercent = value;
        }
    }

    public double TemperatureC
    {
        get => _temperatureC;
        set
        {
            // Диапазон -30..+70 °C
            if (value < -30) _temperatureC = -30;
            else if (value > 70) _temperatureC = 70;
            else _temperatureC = value;
        }
    }

    public int CycleCount
    {
        get => _cycleCount;
        set => _cycleCount = value < 0 ? 0 : value;
    }

    // Вычисляемое свойство только для чтения — статус перегрева
    public bool IsOverheated => _temperatureC > 55;

    // Конструктор по умолчанию (базовые значения)
    public BatteryPack()
    {
        _capacityKwh = 60.0;
        _chargePercent = 80;
        _temperatureC = 20.0;
        _cycleCount = 0;
    }

    // Конструктор со всеми параметрами (с использованием this)
    public BatteryPack(double capacityKwh, int chargePercent, double temperatureC, int cycleCount)
    {
        this.CapacityKwh = capacityKwh;
        this.ChargePercent = chargePercent;
        this.TemperatureC = temperatureC;
        this.CycleCount = cycleCount;
    }

    // Метод изменения состояния: расход заряда (в кВт·ч)
    public void ConsumeCharge(double kwh)
    {
        if (kwh <= 0) return;

        // Переводим расход в проценты от ёмкости
        double percentLoss = (kwh / _capacityKwh) * 100.0;
        ChargePercent -= (int)Math.Round(percentLoss);

        // Небольшой нагрев при интенсивной нагрузке (упрощённо)
        if (kwh > 5.0)
            TemperatureC += 2.0;
    }

    // Метод восстановления: зарядка с контролем переполнения
    public void Recharge(double kwh)
    {
        if (kwh <= 0) return;

        double percentGain = (kwh / _capacityKwh) * 100.0;
        int targetPercent = ChargePercent + (int)Math.Round(percentGain);

        // Ограничиваем максимум 100%
        ChargePercent = Math.Min(100, targetPercent);

        // При зарядке батарея немного нагревается
        TemperatureC += 1.5;

        // Каждый полный цикл зарядки увеличивает счётчик (упрощённо: если заряд был <50 и стал >=80)
        if (_chargePercent < 50 && ChargePercent >= 80)
            _cycleCount++;
    }

    // Метод PrintStatus(): форматированный вывод всех параметров
    public void PrintStatus(string label)
    {
        Console.WriteLine($"[{label}] Ёмкость: {_capacityKwh:F1} кВт·ч | Заряд: {_chargePercent}% | " +
                          $"Температура: {_temperatureC:F1}°C | Циклы: {_cycleCount} | " +
                          $"Перегрев: {(IsOverheated ? "ДА" : "НЕТ")}");
    }
}

class Program
{
    static void Main()
    {
        // Создаём массив из 3 объектов с разными начальными параметрами
        BatteryPack[] batteries = new BatteryPack[3];

        batteries[0] = new BatteryPack();                         // конструктор по умолчанию
        batteries[1] = new BatteryPack(52.0, 90, 15.0, 300);      // новая батарея, почти полная
        batteries[2] = new BatteryPack(48.0, 40, -10.0, 850);     // изношенная, холодный старт

        Console.WriteLine("=== Начальное состояние ===");
        foreach (var b in batteries)
            b.PrintStatus("АКБ");

        // Моделируем 4 шага событийной нагрузки
        Console.WriteLine("\n=== Шаг 1: Поездка в мороз (расход + охлаждение) ===");
        batteries[0].ConsumeCharge(8.0);
        batteries[0].TemperatureC -= 1.0;                        // мороз снижает температуру
        batteries[1].ConsumeCharge(12.0);
        batteries[2].ConsumeCharge(7.0);

        Console.WriteLine("\n=== Шаг 2: Скоростная зарядка (Recharge) ===");
        batteries[0].Recharge(15.0);
        batteries[1].Recharge(20.0);
        batteries[2].Recharge(10.0);

        Console.WriteLine("\n=== Шаг 3: Интенсивный разгон (большой расход) ===");
        batteries[0].ConsumeCharge(20.0);
        batteries[1].ConsumeCharge(18.0);
        batteries[2].ConsumeCharge(15.0);

        Console.WriteLine("\n=== Шаг 4: Длительная стоянка на солнце (нагрев без нагрузки) ===");
        batteries[0].TemperatureC += 5.0;
        batteries[1].TemperatureC += 4.0;
        batteries[2].TemperatureC += 6.0;

        // Итоговое состояние всех объектов
        Console.WriteLine("\n=== Итоговое состояние ===");
        foreach (var b in batteries)
            b.PrintStatus("АКБ");
    }
}
